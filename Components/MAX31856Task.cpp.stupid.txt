/*
 * MAX31856Task.cpp
 *
 * Reads MAX31856 thermocouple samples and prints temperature.
 */

#include <MAX31856MUD+/newdriver.h.txt>
#include <MAX31856Task.hpp>

extern "C" {
#include "main.h"
extern SPI_HandleTypeDef hspi1;
}

namespace {
constexpr int MAX31856_SENSOR_COUNT = 3;

constexpr unsigned long MAX31856_REINIT_PERIOD_MS = 1000;
constexpr unsigned long MAX31856_READ_PERIOD_MS = 1000;
constexpr unsigned long MAX31856_COMMAND_TIMEOUT_MS = 20;

constexpr uint8_t MAX31856_CR0_INIT =
		CR0_AUTOMATIC_CONVERSION
		| CR0_FAULT_INTERRUPT_MODE;

constexpr uint8_t MAX31856_CR1_INIT =
    CR1_THERMOCOUPLE_TYPE_K
    | CR1_AVERAGE_4_SAMPLES;

constexpr uint8_t MAX31856_MASK_INIT = 0x00;

GPIO_TypeDef* const MAX31856_CS_GPIO_PORTS[MAX31856_SENSOR_COUNT] = {
    TC1_cs_GPIO_Port,
    TC2_cs_GPIO_Port,
    TC3_cs_GPIO_Port
};

const uint16_t MAX31856_CS_PINS[MAX31856_SENSOR_COUNT] = {
    static_cast<uint16_t>(TC1_cs_Pin),
    static_cast<uint16_t>(TC2_cs_Pin),
    static_cast<uint16_t>(TC3_cs_Pin)
};
}

MAX31856Task::MAX31856Task()
    : Task(TASK1_QUEUE_DEPTH_OBJS)
{
    for (int i = 0; i < NUM_SENSORS; ++i) {
        _thermos[i] = new MAX31856(
            GPIOA, GPIO_PIN_7,    // SDI (MOSI - PA7)
            GPIOA, GPIO_PIN_6,    // SDO (MISO - PA6)
            MAX31856_CS_GPIO_PORTS[i], MAX31856_CS_PINS[i],  // CS
            GPIOA, GPIO_PIN_5     // CLK (SCK - PA5)
        );
        _enableReading[i] = true;
        _sensorReady[i] = false;
    }
}

bool SendMax31856TaskCommand(uint16_t taskCommand)
{
    Command cmd(DATA_COMMAND, taskCommand);
    return MAX31856Task::Inst().GetEventQueue()->Send(cmd);
}

/**
 * @brief Initializes MAX31856Task with the RTOS scheduler
 */
void MAX31856Task::InitTask()
{
    // Make sure the task is not already initialized
    SOAR_ASSERT(rtTaskHandle == nullptr, "Cannot initialize MAX31856Task twice");

    // Start the task
    BaseType_t rtValue =
        xTaskCreate((TaskFunction_t)MAX31856Task::RunTask,
            (const char*)"MAX31856Task",
            (uint16_t)TASK1_STACK_DEPTH_WORDS,
            (void*)this,
            (UBaseType_t)TASK1_RTOS_PRIORITY,
            (TaskHandle_t*)&rtTaskHandle);

    // Ensure creation
    SOAR_ASSERT(rtValue == pdPASS, "MAX31856Task::InitTask() - xTaskCreate() failed");
}

/**
 * @brief Instance run loop for MAX31856Task.
 * @param pvParams RTOS passed void parameters, unused.
 */
void MAX31856Task::Run(void* pvParams)
{
    (void)pvParams;

    auto initializeSensor = [this](int index) -> bool {
        MAX31856* thermo = _thermos[index];
        
        thermo->writeRegister(REGISTER_CR0, MAX31856_CR0_INIT);
        thermo->writeRegister(REGISTER_CR1, MAX31856_CR1_INIT);
        thermo->writeRegister(REGISTER_MASK, MAX31856_MASK_INIT);

        SOAR_PRINT("MAX31856Task - TC%d initialized\n", index + 1);
        SOAR_PRINT("MAX31856Task - TC%d registers configured (CR0=0x%02X CR1=0x%02X MASK=0x%02X)\n",
                   index + 1,
                   static_cast<unsigned int>(MAX31856_CR0_INIT),
                   static_cast<unsigned int>(MAX31856_CR1_INIT),
                   static_cast<unsigned int>(MAX31856_MASK_INIT));

        return true;
    };

    auto readSensor = [this](int index) {
        if (!_sensorReady[index] || !_enableReading[index]) {
            return;
        }

        const double tempC = _thermos[index]->readThermocouple(CELSIUS);
        if (tempC == FAULT_OPEN) {
            SOAR_PRINT("MAX31856Task - TC%d fault: OPEN\n", index + 1);
        } else if (tempC == FAULT_VOLTAGE) {
            SOAR_PRINT("MAX31856Task - TC%d fault: VOLTAGE\n", index + 1);
        } else if (tempC == NO_MAX31856) {
            SOAR_PRINT("MAX31856Task - TC%d fault: NO_DEVICE\n", index + 1);
        } else {
            SOAR_PRINT("MAX31856Task - TC%d temp: %.2f C\n", index + 1, tempC);
        }
    };

    TickType_t lastInitRetryTick[NUM_SENSORS];
    TickType_t lastReadTick[NUM_SENSORS];
    const TickType_t initStartTick = xTaskGetTickCount();
    for (int i = 0; i < NUM_SENSORS; ++i) {
        lastInitRetryTick[i] = initStartTick;
        lastReadTick[i] = initStartTick;
        _sensorReady[i] = initializeSensor(i);
        if (_sensorReady[i]) {
            readSensor(i);
        }
    }

    while (1) {
        Command cm;
        if (qEvtQueue->Receive(cm, MAX31856_COMMAND_TIMEOUT_MS)) {
            HandleCommand(cm);
            continue;
        }

        const TickType_t now = xTaskGetTickCount();

        for (int i = 0; i < NUM_SENSORS; ++i) {
            if (!_sensorReady[i]) {
                if ((now - lastInitRetryTick[i]) >= pdMS_TO_TICKS(MAX31856_REINIT_PERIOD_MS)) {
                    lastInitRetryTick[i] = now;
                    _sensorReady[i] = initializeSensor(i);
                    if (_sensorReady[i]) {
                        readSensor(i);
                        lastReadTick[i] = now;
                    }
                }
                continue;
            }

            if (_enableReading[i]
                && ((now - lastReadTick[i]) >= pdMS_TO_TICKS(MAX31856_READ_PERIOD_MS))) {
                lastReadTick[i] = now;
                readSensor(i);
            }
        }
    }
}

/**
 * @brief HandleCommand handles any command passed to MAX31856Task primary event queue.
 * @param cm Reference to the command object to handle
 */
void MAX31856Task::HandleCommand(Command& cm)
{
    switch (cm.GetCommand()) {
    case DATA_COMMAND: {
        switch (cm.GetTaskCommand()) {
        case MAX31856_TASK_COMMAND_TOGGLE:
            for (int i = 0; i < NUM_SENSORS; ++i) {
                _enableReading[i] = !_enableReading[i];
            }
            SOAR_PRINT("MAX31856Task - MAX31856 reading toggled for all sensors\n");
            break;
        case MAX31856_TASK_COMMAND_ON:
            for (int i = 0; i < NUM_SENSORS; ++i) {
                _enableReading[i] = true;
            }
            SOAR_PRINT("MAX31856Task - MAX31856 reading ON (all)\n");
            break;
        case MAX31856_TASK_COMMAND_OFF:
            for (int i = 0; i < NUM_SENSORS; ++i) {
                _enableReading[i] = false;
            }
            SOAR_PRINT("MAX31856Task - MAX31856 reading OFF (all)\n");
            break;
        case MAX31856_TASK_COMMAND_STATUS:
            for (int i = 0; i < NUM_SENSORS; ++i) {
                SOAR_PRINT("MAX31856Task - TC%d read %s, ready=%s\n",
                           i + 1,
                           _enableReading[i] ? "ON" : "OFF",
                           _sensorReady[i] ? "yes" : "no");
            }
            break;
        case MAX31856_TASK_COMMAND_ENABLE_LOG:
            SOAR_PRINT("MAX31856Task - MAX31856 logging control removed; reads stay enabled\n");
            break;
        case MAX31856_TASK_COMMAND_DISABLE_LOG:
            SOAR_PRINT("MAX31856Task - MAX31856 logging control removed; reads stay enabled\n");
            break;
        case MAX31856_TASK_COMMAND_READ_TC1:
        case MAX31856_TASK_COMMAND_READ_TC2:
        case MAX31856_TASK_COMMAND_READ_TC3: {
            const int tcIndex = cm.GetTaskCommand() - MAX31856_TASK_COMMAND_READ_TC1;
            if (_sensorReady[tcIndex] && _enableReading[tcIndex]) {
                const double tempC = _thermos[tcIndex]->readThermocouple(CELSIUS);
                if (tempC == FAULT_OPEN) {
                    SOAR_PRINT("MAX31856Task - TC%d fault: OPEN\n", tcIndex + 1);
                } else if (tempC == FAULT_VOLTAGE) {
                    SOAR_PRINT("MAX31856Task - TC%d fault: VOLTAGE\n", tcIndex + 1);
                } else if (tempC == NO_MAX31856) {
                    SOAR_PRINT("MAX31856Task - TC%d fault: NO_DEVICE\n", tcIndex + 1);
                } else {
                    SOAR_PRINT("MAX31856Task - TC%d temp: %.2f C\n", tcIndex + 1, tempC);
                }
            } else {
                SOAR_PRINT("MAX31856Task - TC%d read skipped (disabled or not initialized)\n", tcIndex + 1);
            }
            break;
        }
        case MAX31856_TASK_COMMAND_READ_FORCE_TC1: {
            if (_sensorReady[0]) {
                const double tempC = _thermos[0]->readThermocouple(CELSIUS);
                if (tempC == FAULT_OPEN) {
                    SOAR_PRINT("MAX31856Task - TC1 forced fault: OPEN\n");
                } else if (tempC == FAULT_VOLTAGE) {
                    SOAR_PRINT("MAX31856Task - TC1 forced fault: VOLTAGE\n");
                } else if (tempC == NO_MAX31856) {
                    SOAR_PRINT("MAX31856Task - TC1 forced fault: NO_DEVICE\n");
                } else {
                    SOAR_PRINT("MAX31856Task - TC1 forced temp: %.2f C\n", tempC);
                }
            } else {
                SOAR_PRINT("MAX31856Task - TC1 force read skipped (not initialized)\n");
            }
            break;
        }
        case MAX31856_TASK_COMMAND_READ_REGS_TC1: {
            SOAR_PRINT("MAX31856Task - TC1 regs configured: CR0=0x%02X CR1=0x%02X MASK=0x%02X\n",
                       static_cast<unsigned int>(MAX31856_CR0_INIT),
                       static_cast<unsigned int>(MAX31856_CR1_INIT),
                       static_cast<unsigned int>(MAX31856_MASK_INIT));
            break;
        }
        case MAX31856_TASK_COMMAND_TOGGLE_CS_TC1: {
            SOAR_PRINT("MAX31856Task - TC1 CS toggle removed\n");
            break;
        }
        case MAX31856_TASK_COMMAND_TOGGLE_TC1:
        case MAX31856_TASK_COMMAND_TOGGLE_TC2:
        case MAX31856_TASK_COMMAND_TOGGLE_TC3: {
            const int tcIndex = cm.GetTaskCommand() - MAX31856_TASK_COMMAND_TOGGLE_TC1;
            _enableReading[tcIndex] = !_enableReading[tcIndex];
            SOAR_PRINT("MAX31856Task - TC%d read %s\n",
                       tcIndex + 1,
                       _enableReading[tcIndex] ? "ON" : "OFF");
            break;
        }
        default:
            SOAR_PRINT("MAX31856Task - Received Unsupported DATA_COMMAND {%d}\n", cm.GetTaskCommand());
            break;
        }
        break;
    }
    default:
        SOAR_PRINT("MAX31856Task - Received Unsupported Command {%d}\n", cm.GetCommand());
        break;
    }

    cm.Reset();
}

extern "C" void MAX31856Task_HandleDrdyInterrupt(uint16_t gpioPin)
{
    Command cmd(DATA_COMMAND, MAX31856_TASK_COMMAND_NONE);
    bool shouldSend = false;

    if (gpioPin == TC1_nReady_Pin) {
        cmd.SetTaskCommand(MAX31856_TASK_COMMAND_READ_TC1);
        shouldSend = true;
    } else if (gpioPin == TC2_nReady_Pin) {
        cmd.SetTaskCommand(MAX31856_TASK_COMMAND_READ_TC2);
        shouldSend = true;
    } else if (gpioPin == TC3_nReady_Pin) {
        cmd.SetTaskCommand(MAX31856_TASK_COMMAND_READ_TC3);
        shouldSend = true;
    }

    if (shouldSend) {
        MAX31856Task::Inst().GetEventQueue()->SendFromISR(cmd);
    }
}
